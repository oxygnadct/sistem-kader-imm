// Mengganti fungsi loadData() async function loadData() { showLoading(true); try { const response = await fetch('/api/kader'); const data = await response.json(); if(data.error) throw new Error(data.error); db = data; // Data dari Turso renderUI(); } catch (e) { Swal.fire("Sistem Error", "Gagal memuat data dari database: " + e.message, "error"); db = { kader: [], config: {} }; renderUI(); } finally { showLoading(false); } } // Mengganti blok pengiriman pada fungsi submitForm(e) async function submitForm(e) { e.preventDefault(); // ... (kode pengambilan payload sama seperti sebelumnya) ... const payload = { id: document.getElementById('formId').value, nama: document.getElementById('nama').value, // ... (lanjutkan payload sesuai kode aslinya) }; const isEdit = payload.id !== ''; const btn = document.getElementById('btnSubmitForm'); const originalBtnHtml = btn.innerHTML; btn.disabled = true; btn.innerHTML = ' Menyimpan...'; try { const response = await fetch('/api/kader', { method: isEdit ? 'PUT' : 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) }); const res = await response.json(); if(res.success) { closeModal(); showToast(res.message, 'success'); loadData(); // Memuat ulang data dari Turso } else { Swal.fire("Gagal", res.message, "error"); } } catch (err) { Swal.fire("Error Server", err.message, "error"); } finally { btn.disabled = false; btn.innerHTML = originalBtnHtml; } } // Mengganti pemanggilan API pada fungsi deleteKader(id) function deleteKader(id) { Swal.fire({ title: 'Apakah Anda Yakin?', text: "Data kader ini akan dihapus permanen!", icon: 'warning', showCancelButton: true, confirmButtonColor: '#c92a2a', cancelButtonColor: '#6b7280', confirmButtonText: 'Ya, Hapus!' }).then(async (result) => { if (result.isConfirmed) { showLoading(true); try { const response = await fetch(`/api/kader?id=${id}`, { method: 'DELETE' }); const res = await response.json(); if(res.success) { showToast(res.message, 'success'); loadData(); } else { Swal.fire("Gagal", res.message, "error"); showLoading(false); } } catch(err) { Swal.fire("Error", err.message, "error"); showLoading(false); } } }); }
